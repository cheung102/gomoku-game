### 4.1 创建 game.py（控制台版本）

核心类设计：

```python
class Board:
    """棋盘类"""
    def __init__(self, size=15):
        self.size = size
        self.grid = [[0 for _ in range(size)] for _ in range(size)]
    
    def is_valid_move(self, row, col):
        """检查落子是否有效"""
        return (0 <= row < self.size and 
                0 <= col < self.size and 
                self.grid[row][col] == 0)
    
    def make_move(self, row, col, player):
        """执行落子"""
        if self.is_valid_move(row, col):
            self.grid[row][col] = player
            return True
        return False
    
    def check_win(self, player):
        """检查是否获胜 - 四个方向扫描"""
        directions = [(0, 1), (1, 0), (1, 1), (1, -1)]
        for row in range(self.size):
            for col in range(self.size):
                if self.grid[row][col] == player:
                    for dr, dc in directions:
                        count = 1
                        for i in range(1, 5):
                            r, c = row + dr * i, col + dc * i
                            if (0 <= r < self.size and 
                                0 <= c < self.size and 
                                self.grid[r][c] == player):
                                count += 1
                            else:
                                break
                        if count >= 5:
                            return True
        return False
    
    def is_full(self):
        """检查棋盘是否已满"""
        for row in range(self.size):
            for col in range(self.size):
                if self.grid[row][col] == 0:
                    return False
        return True
```

```python
def draw_board(self):
```

```python
            if self.board.grid[row][col] != 0:
```

### 5.4 鼠标坐标转换

```python
def get_grid_pos(self, mouse_pos):
    """将鼠标像素坐标转换为棋盘行列坐标"""
    x, y = mouse_pos
    # 四舍五入到最近的交叉点
    col = round((x - MARGIN) / CELL_SIZE)
    row = round((y - MARGIN) / CELL_SIZE)
    if 0 <= row < BOARD_SIZE and 0 <= col < BOARD_SIZE:
        return row, col
    return None
```

### 6.1 创建音效生成脚本

创建 `generate_sounds.py`，使用 Python 内置模块生成简单音效：

```python
import wave
import struct
import math

def generate_sine_wave(frequency, duration, sample_rate=44100, amplitude=0.5):
    """生成正弦波"""
    num_samples = int(sample_rate * duration)
    samples = []
    for i in range(num_samples):
        t = i / sample_rate
        value = amplitude * math.sin(2 * math.pi * frequency * t)
        samples.append(value)
    return samples

def save_wav(filename, samples, sample_rate=44100):
    """保存为 WAV 文件"""
    with wave.open(filename, 'w') as wav_file:
        wav_file.setnchannels(1)
        wav_file.setsampwidth(2)
        wav_file.setframerate(sample_rate)
        # 将浮点样本转换为16位整数
        max_sample = max(abs(s) for s in samples) if samples else 1
        if max_sample > 0:
            samples = [int(s / max_sample * 32767) for s in samples]
        wav_file.writeframes(struct.pack(f'<{len(samples)}h', *samples))
```

生成三种音效：
- `sounds/place.wav` - 落子音效（短促点击声）

### 6.2 添加落子动画

创建 `PieceAnimation` 类实现棋子缩放动画：

```python
class PieceAnimation:
    def __init__(self, row, col, player, start_time):
        self.row = row
        self.col = col
        self.player = player
        self.start_time = start_time
        self.duration = 200  # 动画持续200毫秒
        self.scale = 0.0     # 初始缩放为0
    
    def update(self, current_time):
        """更新动画状态"""
        elapsed = current_time - self.start_time
        if elapsed < self.duration:
            self.scale = elapsed / self.duration  # 0.0 -> 1.0
            return True
        else:
            self.scale = 1.0
            return False
    
    def draw(self, surface, cell_size, margin):
        """绘制动画中的棋子"""
        if self.scale <= 0:
            return
        center_x = margin + self.col * cell_size
        center_y = margin + self.row * cell_size
        radius = (cell_size // 2 - 2) * self.scale
        color = BLACK if self.player == 1 else WHITE
        pygame.draw.circle(surface, color, (center_x, center_y), int(radius))
```

## 7. AI 对手开发

### 7.1 AI 类设计

创建 `GomokuAI` 类实现简单的人工智能：

```python
class GomokuAI:
    """简单AI玩家"""
    def __init__(self, player=2):
        self.player = player      # AI执白棋
        self.opponent = 1 if player == 2 else 2
    
    def get_move(self, board):
        """获取AI落子位置"""
        # 1. 检查AI是否能赢（优先级最高）
        win_move = self._find_winning_move(board, self.player)
        if win_move:
            return win_move
        
        # 2. 检查对手是否能赢，进行阻挡
        block_move = self._find_winning_move(board, self.opponent)
        if block_move:
            return block_move
        
        # 3. 寻找最佳进攻位置（基于评分）
        best_move = self._find_best_move(board)
        if best_move:
            return best_move
        
        # 4. 如果棋盘为空，下在中心
        center = board.size // 2
        if board.grid[center][center] == 0:
            return (center, center)
        
        # 5. 随机找一个空位
        return self._find_random_move(board)
```

### 7.2 位置评估函数

AI 通过评分系统选择最佳位置：

```python
def _evaluate_position(self, board, row, col):
    """评估某个位置的价值"""
    score = 0
    directions = [(0, 1), (1, 0), (1, 1), (1, -1)]
    
    for dr, dc in directions:
        for direction in [1, -1]:
            count = 0
            empty = 0
            for i in range(1, 5):
                r, c = row + dr * i * direction, col + dc * i * direction
                if 0 <= r < board.size and 0 <= c < board.size:
                    if board.grid[r][c] == self.player:
                        count += 1
                    elif board.grid[r][c] == 0:
                        empty += 1
                        break
                    else:
                        break
            
            # 评分规则
            if count >= 4:
                score += 10000  # 活四（必赢）
            elif count == 3 and empty > 0:
                score += 1000   # 活三
            elif count == 2 and empty > 0:
                score += 100    # 活二
            elif count == 1 and empty > 0:
                score += 10     # 活一
    
    # 中心位置加分
    center = board.size // 2
    distance = abs(row - center) + abs(col - center)
    score += max(0, 15 - distance)
    
    return score
```

```python
# tests/test_board.py
import unittest
from game import Board

class TestBoard(unittest.TestCase):
    def setUp(self):
        self.board = Board(15)
    
    def test_initial_board(self):
        """测试棋盘初始化"""
        self.assertEqual(self.board.size, 15)
        for row in range(15):
            for col in range(15):
                self.assertEqual(self.board.grid[row][col], 0)
    
    def test_valid_move(self):
        """测试有效落子"""
        self.assertTrue(self.board.make_move(7, 7, 1))
        self.assertEqual(self.board.grid[7][7], 1)
    
    def test_check_win_horizontal(self):
        """测试水平方向胜利"""
        for i in range(5):
            self.board.make_move(7, 7 + i, 1)
        self.assertTrue(self.board.check_win(1))
    
    def test_check_win_vertical(self):
        """测试垂直方向胜利"""
        for i in range(5):
            self.board.make_move(7 + i, 7, 1)
        self.assertTrue(self.board.check_win(1))
    
    def test_check_win_diagonal(self):
        """测试对角线方向胜利"""
        for i in range(5):
            self.board.make_move(7 + i, 7 + i, 1)
        self.assertTrue(self.board.check_win(1))
    
    def test_no_win_incomplete(self):
        """测试未完成五子连珠"""
        for i in range(4):
            self.board.make_move(7, 7 + i, 1)
        self.assertFalse(self.board.check_win(1))
```

**问题**：pygame 默认字体不支持中文，导致界面文字显示为方块。

**解决方案**：加载系统中文字体

```python
def _load_chinese_font(self, size):
    """加载系统中文字体"""
    font_paths = [
        "C:/Windows/Fonts/simhei.ttf",    # 黑体
        "C:/Windows/Fonts/simsun.ttc",    # 宋体
        "C:/Windows/Fonts/msyh.ttc",      # 微软雅黑
    ]
    
    for font_path in font_paths:
        if os.path.exists(font_path):
            try:
                return pygame.font.Font(font_path, size)
            except:

    return pygame.font.Font(None, size)  # 回退到默认字体
```

### 9.2 鼠标坐标偏移问题

**问题**：鼠标点击位置与实际棋子落点不一致。

**原因**：坐标计算使用整数除法导致偏移。

**解决方案**：改用四舍五入

```python
def get_grid_pos(self, mouse_pos):
    x, y = mouse_pos
    # 四舍五入到最近的交叉点
    col = round((x - MARGIN) / CELL_SIZE)
    row = round((y - MARGIN) / CELL_SIZE)
    if 0 <= row < BOARD_SIZE and 0 <= col < BOARD_SIZE:
        return row, col
    return None
```

同时修正边距计算：

```python
# 修正前

```
