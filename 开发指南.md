## 4. 核心代码开发

### 4.1 创建 game.py（控制台版本）

核心类设计：

```python
class Board:
    """棋盘类"""
    def __init__(self, size=15):
        self.size = size
        self.grid = [[0 for _ in range(size)] for _ in range(size)]
        self.move_history = []
    
    def is_valid_move(self, row, col):
        """检查落子是否有效"""
        return (0 <= row < self.size and 
                0 <= col < self.size and 
                self.grid[row][col] == 0)
    
    def make_move(self, row, col, player):
        """执行落子"""
        if self.is_valid_move(row, col):
            self.grid[row][col] = player
            self.move_history.append((row, col, player))
            return True
        return False
    
    def undo_move(self):
        """悔棋"""
        if self.move_history:
            row, col, _ = self.move_history.pop()
            self.grid[row][col] = 0
            return True
        return False
    
    def check_win(self, player):
        """检查是否获胜"""
        return self.get_winning_line(player) is not None
    
    def get_winning_line(self, player):
        """获取获胜的五子连线"""
        directions = [(0, 1), (1, 0), (1, 1), (1, -1)]
        for row in range(self.size):
            for col in range(self.size):
                if self.grid[row][col] == player:
                    for dr, dc in directions:
                        line = [(row, col)]
                        for i in range(1, 5):
                            r, c = row + dr * i, col + dc * i
                            if (0 <= r < self.size and 
                                0 <= c < self.size and 
                                self.grid[r][c] == player):
                                line.append((r, c))
                            else:
                                break
                        if len(line) == 5:
                            return line
        return None
    
    def is_full(self):
        """检查棋盘是否已满"""
        for row in range(self.size):
            for col in range(self.size):
                if self.grid[row][col] == 0:
                    return False
        return True
```

关键方法说明：
- `make_move()`：落子并记录历史
- `undo_move()`：悔棋，移除最后一步
- `get_winning_line()`：返回获胜的五子坐标列表
- `check_win()`：检查是否获胜

class PieceAnimation:
    """棋子放置动画"""
    def __init__(self, row, col, player, start_time):
        self.row = row
        self.col = col
        self.player = player
        self.start_time = start_time
        self.duration = 100  # 毫秒
        self.progress = 0
    
    def update(self, current_time):
        elapsed = current_time - self.start_time
        self.progress = min(1.0, elapsed / self.duration)
    
    def is_complete(self):
        return self.progress >= 1.0


class WinningLineAnimation:
    """获胜连线动画"""
    def __init__(self, winning_line, start_time):
        self.line = winning_line
        self.start_time = start_time
        self.duration = 500  # 毫秒
        self.progress = 0
    
    def update(self, current_time):
        elapsed = current_time - self.start_time
        self.progress = min(1.0, elapsed / self.duration)


class HoverPreview:
    """鼠标悬停预览"""
    def __init__(self):
        self.position = None
        self.visible = False
    
    def update(self, mouse_pos, board, cell_size, margin, board_size):
        col = round((mouse_pos[0] - margin) / cell_size)
        row = round((mouse_pos[1] - margin) / cell_size)
        if 0 <= row < board_size and 0 <= col < board_size:
            if board.is_valid_move(row, col):
                self.position = (row, col)
                self.visible = True
                return
        self.visible = False

    """最后落子标记"""
    def __init__(self, row, col):
        self.row = row
        self.col = col

class GomokuAI:
    """五子棋 AI 玩家"""
    
    def __init__(self, player=2, difficulty='medium'):
        self.player = player
        self.opponent = 1 if player == 2 else 2
        self.difficulty = difficulty
        self.search_depth = {'easy': 1, 'medium': 2, 'hard': 3}.get(difficulty, 2)
    
    def get_move(self, board):

        if self.difficulty == 'easy':

        elif self.difficulty == 'medium':

        else:
            return self._hard_move(board)

### 7.2 难度说明

| 难度 | 搜索深度 | 候选数 | 说明 |
|------|----------|--------|------|
| 简单 | 1 | 全部 | 随机落子 + 基础阻挡 |
| 中等 | 2 | 10 | 评分系统选择最佳位置 |
| 困难 | 3 | 10 | Minimax + Alpha-Beta 剪枝 |

### 7.3 AI 核心算法

```python
def _minimax(self, board, depth, alpha, beta, is_maximizing):
    """Minimax + Alpha-Beta 剪枝算法"""
    self.node_count += 1
    if board.check_win(self.player):
        return 100000 + depth
    if board.check_win(self.opponent):
        return -100000 - depth
    if depth == 0 or board.is_full():
        return self._evaluate_board(board)
    
    candidate_moves = self._get_candidate_moves(board)
    if len(candidate_moves) > 10:
        candidate_moves = self._pre_sort_moves(board, candidate_moves)[:10]
    
    if is_maximizing:
        max_eval = -float('inf')
        for row, col in candidate_moves:
            board.make_move(row, col, self.player)
            eval_score = self._minimax(board, depth - 1, alpha, beta, False)
            board.undo_move()
            max_eval = max(max_eval, eval_score)
            alpha = max(alpha, eval_score)
            if beta <= alpha:
                break
        return max_eval
    else:
        min_eval = float('inf')
        for row, col in candidate_moves:
            board.make_move(row, col, self.opponent)
            eval_score = self._minimax(board, depth - 1, alpha, beta, True)
            board.undo_move()
            min_eval = min(min_eval, eval_score)
            beta = min(beta, eval_score)
            if beta <= alpha:
                break
        return min_eval
```

### 7.4 提示功能

```python
def get_hint_positions(self, board, num_hints=3):
    """获取 AI 推荐的落子位置"""
    candidates = self._get_candidate_moves(board)
    if not candidates:
        return [(board.size // 2, board.size // 2)]
    
    scored_moves = []
    for row, col in candidates:
        board.make_move(row, col, self.player)
        score = self._evaluate_board(board)
        board.undo_move()
        scored_moves.append(((row, col), score))
    
    scored_moves.sort(key=lambda x: x[1], reverse=True)
    return [move for move, score in scored_moves[:num_hints]]
```

import sys

class TestBoard(unittest.TestCase):

        """测试水平方向胜利"""

**问题描述**：两个音效同时播放，声音重叠。

```python
def play_sound(self, sound_name):

        self.sounds[sound_name].play()

## 19. 常见问题解决

# 1. 发生冲突时

```

### 19.2 合并策略

| 策略 | 命令 | 优点 | 缺点 |
|------|------|------|------|
| 快进合并 | `git merge feature` | 简单快速 | 丢失分支历史 |
| 普通合并 | `git merge --no-ff feature` | 保留分支历史 | 多一个合并提交 |
| 变基合并 | `git rebase develop` | 线性历史 | 需要 force push 风险 |
| 压缩合并 | `git merge --squash feature` | 提交历史整洁 | 丢失详细历史 |

### 19.3 撤销操作

```powershell
# 撤销工作区修改
git checkout -- <file>

# 撤销暂存区修改
git reset HEAD <file>

# 撤销最近一次提交（保留修改）
git reset --soft HEAD~1

# 撤销最近一次提交（丢弃修改）
git reset --hard HEAD~1
```

### 19.4 查看历史
