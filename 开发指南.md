### 4.1 创建 game.py（控制台版本）

核心类设计：

```python
class Board:
    """棋盘类"""
    def __init__(self, size=15):
        self.size = size
        self.grid = [[0 for _ in range(size)] for _ in range(size)]
    
    def is_valid_move(self, row, col):
        """检查落子是否有效"""
        return (0 <= row < self.size and 
                0 <= col < self.size and 
                self.grid[row][col] == 0)
    
    def make_move(self, row, col, player):
        """执行落子"""
        if self.is_valid_move(row, col):
            self.grid[row][col] = player
            return True
        return False
    
    def check_win(self, player):
        """检查是否获胜 - 四个方向扫描"""
        directions = [(0, 1), (1, 0), (1, 1), (1, -1)]
        for row in range(self.size):
            for col in range(self.size):
                if self.grid[row][col] == player:
                    for dr, dc in directions:
                        count = 1
                        for i in range(1, 5):
                            r, c = row + dr * i, col + dc * i
                            if (0 <= r < self.size and 
                                0 <= c < self.size and 
                                self.grid[r][c] == player):
                                count += 1
                            else:
                                break
                        if count >= 5:
                            return True
        return False
    
    def is_full(self):
        """检查棋盘是否已满"""
        for row in range(self.size):
            for col in range(self.size):
                if self.grid[row][col] == 0:
                    return False
        return True
```

```python
def get_grid_pos(self, mouse_pos):
    """将鼠标像素坐标转换为棋盘行列坐标"""
    x, y = mouse_pos
    # 四舍五入到最近的交叉点

    return None
```

### 6.1 创建音效生成脚本

创建 `generate_sounds.py`，使用 Python 内置模块生成简单音效：

```python
import wave
import struct
import math

def generate_sine_wave(frequency, duration, sample_rate=44100, amplitude=0.5):
    """生成正弦波"""
    num_samples = int(sample_rate * duration)
    samples = []
    for i in range(num_samples):
        t = i / sample_rate
        value = amplitude * math.sin(2 * math.pi * frequency * t)
        samples.append(value)
    return samples

def save_wav(filename, samples, sample_rate=44100):
    """保存为 WAV 文件"""
    with wave.open(filename, 'w') as wav_file:
        wav_file.setnchannels(1)
        wav_file.setsampwidth(2)
        wav_file.setframerate(sample_rate)
        # 将浮点样本转换为16位整数
        max_sample = max(abs(s) for s in samples) if samples else 1
        if max_sample > 0:
            samples = [int(s / max_sample * 32767) for s in samples]
        wav_file.writeframes(struct.pack(f'<{len(samples)}h', *samples))
```

### 6.2 添加落子动画

创建 `PieceAnimation` 类实现棋子缩放动画：

```python
class PieceAnimation:
    def __init__(self, row, col, player, start_time):
        self.row = row
        self.col = col
        self.player = player
        self.start_time = start_time
        self.duration = 200  # 动画持续200毫秒
        self.scale = 0.0     # 初始缩放为0
    
    def update(self, current_time):
        """更新动画状态"""
        elapsed = current_time - self.start_time
        if elapsed < self.duration:
            self.scale = elapsed / self.duration  # 0.0 -> 1.0
            return True
        else:
            self.scale = 1.0
            return False
    
    def draw(self, surface, cell_size, margin):
        """绘制动画中的棋子"""
        if self.scale <= 0:
            return
        center_x = margin + self.col * cell_size
        center_y = margin + self.row * cell_size
        radius = (cell_size // 2 - 2) * self.scale
        color = BLACK if self.player == 1 else WHITE
        pygame.draw.circle(surface, color, (center_x, center_y), int(radius))
```

## 7. AI 对手开发

### 7.1 AI 类设计

创建 `GomokuAI` 类实现简单的人工智能：

```python
class GomokuAI:
    """简单AI玩家"""
    def __init__(self, player=2):
        self.player = player      # AI执白棋
        self.opponent = 1 if player == 2 else 2
    
    def get_move(self, board):
        """获取AI落子位置"""
        # 1. 检查AI是否能赢（优先级最高）
        win_move = self._find_winning_move(board, self.player)
        if win_move:
            return win_move
        
        # 2. 检查对手是否能赢，进行阻挡
        block_move = self._find_winning_move(board, self.opponent)
        if block_move:
            return block_move
        
        # 3. 寻找最佳进攻位置（基于评分）
        best_move = self._find_best_move(board)
        if best_move:
            return best_move
        
        # 4. 如果棋盘为空，下在中心
        center = board.size // 2
        if board.grid[center][center] == 0:
            return (center, center)
        
        # 5. 随机找一个空位
        return self._find_random_move(board)
```

### 7.2 位置评估函数

AI 通过评分系统选择最佳位置：

```python
def _evaluate_position(self, board, row, col):
    """评估某个位置的价值"""
    score = 0
    directions = [(0, 1), (1, 0), (1, 1), (1, -1)]
    
    for dr, dc in directions:
        for direction in [1, -1]:
            count = 0
            empty = 0
            for i in range(1, 5):
                r, c = row + dr * i * direction, col + dc * i * direction
                if 0 <= r < board.size and 0 <= c < board.size:
                    if board.grid[r][c] == self.player:
                        count += 1
                    elif board.grid[r][c] == 0:
                        empty += 1
                        break
                    else:
                        break
            
            # 评分规则
            if count >= 4:
                score += 10000  # 活四（必赢）
            elif count == 3 and empty > 0:
                score += 1000   # 活三
            elif count == 2 and empty > 0:
                score += 100    # 活二
            elif count == 1 and empty > 0:
                score += 10     # 活一
    
    # 中心位置加分
    center = board.size // 2
    distance = abs(row - center) + abs(col - center)
    score += max(0, 15 - distance)
    
    return score
```

```python
import unittest
from game import Board

class TestBoard(unittest.TestCase):
    def setUp(self):
        self.board = Board(15)

        """测试棋盘初始化"""

        """测试水平方向胜利"""
        for i in range(5):
            self.board.make_move(7, 7 + i, 1)
        self.assertTrue(self.board.check_win(1))
    
    def test_check_win_vertical(self):
        """测试垂直方向胜利"""
        for i in range(5):
            self.board.make_move(7 + i, 7, 1)
        self.assertTrue(self.board.check_win(1))

        """测试对角线方向胜利"""
```

### 9.2 鼠标坐标偏移问题

**问题**：鼠标点击位置与实际棋子落点不一致。

**原因**：坐标计算使用整数除法导致偏移。

**解决方案**：改用四舍五入

```python
def get_grid_pos(self, mouse_pos):
    x, y = mouse_pos
    # 四舍五入到最近的交叉点
    col = round((x - MARGIN) / CELL_SIZE)
    row = round((y - MARGIN) / CELL_SIZE)
    if 0 <= row < BOARD_SIZE and 0 <= col < BOARD_SIZE:
        return row, col
    return None
```

## 12. Git Flow 工作流

经典的分支管理模型，适合有明确发布周期的大型项目。

### 1.1 分支结构

```
master ─────────●──────────────●──────────────●─── (生产环境)
                 \            /              /
                  ●────●────●──────────────●──── (发布分支 release)
                 /        \              /
develop ──●────●────●────●────●────●────●─────── (开发主线)
           \        /        \        /
            ●────●            ●────●             (功能分支 feature)
```

### 1.2 五种分支类型

| 分支 | 命名 | 生命周期 | 用途 |
|------|------|----------|------|
| master | `master` | 永久 | 生产环境代码，只接受 release 和 hotfix 合并 |
| develop | `develop` | 永久 | 开发主线，集成所有功能 |
| feature | `feature/*` | 临时 | 开发新功能，完成后合并到 develop |
| release | `release/*` | 临时 | 准备发布，只做 bug 修复和文档更新 |
| hotfix | `hotfix/*` | 临时 | 紧急修复生产环境 bug |

### 1.3 常用命令

```powershell
# 创建功能分支
git checkout develop
git checkout -b feature/ai-player

# 开发完成，合并回 develop
git checkout develop
git merge feature/ai-player

# 删除已合并的功能分支
git branch -d feature/ai-player

# 推送到远程
git push origin develop
```

### 1.4 优劣势

| 优势 | 劣势 |
|------|------|
| 职责清晰，代码流向明确 | 分支多，流程复杂 |
| 支持并行发布多个版本 | 合并频繁，学习曲线陡 |
| 风险可控，release 隔离发布 | 不适合持续部署 |
| 适合大团队协作 | 维护成本高 |

## 13. Git Flow Topic 工作流

Git Flow 的简化版本，去掉 release 和 hotfix，只保留核心分支。

### 2.1 分支结构

```
main ─────────────●────────────────────●────── (发布版本)
                   \                  /
develop ──●────●────●────●────●────●────────── (开发主线)
           \        /        \        /
            ●────●            ●────●          (topic 分支)
           feature/          feature/
           login             ai-player
```

### 2.2 分支类型

| 分支 | 命名 | 用途 |
|------|------|------|
| main | `main` | 生产环境代码 |
| develop | `develop` | 开发主线 |
| topic | `topic/*` 或 `feature/*` | 功能开发 |

### 2.3 git-flow 扩展工具

```powershell
# 安装 git-flow
sudo apt install git-flow        # Linux
brew install git-flow             # macOS
choco install gitflow            # Windows

# 初始化
git flow init

# 创建 topic 分支
git flow topic start new-feature

# 完成并合并（自动合并到 develop 并删除 topic 分支）
git flow topic finish new-feature

# 查看所有 topic 分支
git flow topic list
```

### 2.4 常用命令

```powershell
# 1. 开始新功能
git flow topic start feature-ai

# 2. 开发并提交
git add .
git commit -m "Add AI player"

# 3. 完成并合并
git flow topic finish feature-ai

# 4. 发布时合并到 main
git checkout main
git merge --no-ff develop
git tag -a v1.0.0 -m "Release version 1.0.0"
```

### 2.5 优劣势

| 优势 | 劣势 |
|------|------|
| 简单易学，只有三种分支 | 缺少发布管理，版本靠 tag |
| 快速迭代，适合敏捷开发 | 不适合多版本并行维护 |
| topic 分支生命周期短，冲突少 | hotfix 没有固定流程 |
| 灵活轻量，适合小团队 | develop 分支压力大 |

## 14. GitHub Flow 工作流

最简单的工作流，只有 main + feature 分支，通过 Pull Request 合并。

### 3.1 分支结构

```
main ────●────●────●────●────── (生产分支)
           \      /        \
            ●────●          ●────●            (功能分支)
           feature/        feature/
```

### 3.2 工作流程

```powershell
# 1. 从 main 创建分支
git checkout main
git checkout -b feature/ai-player

# 2. 开发并提交
git add .
git commit -m "Add AI player"

# 3. 推送到远程
git push origin feature/ai-player

# 4. 在 GitHub 上创建 Pull Request
#    → 代码审查 → 合并到 main → 删除分支
```

### 3.3 为什么 GitHub Flow 最流行

| 原因 | 说明 |
|------|------|
| 简单 | 只有两个分支，概念清晰 |
| 适配 CI/CD | 自动化测试 + 自动部署 |
| 平台支持 | GitHub/GitLab/Bitbucket 原生支持 PR |
| 社区主流 | 开源项目和公司都在用 |

### 3.4 优劣势

| 优势 | 劣势 |
|------|------|
| 极其简单，新手友好 | 不适合版本发布管理 |
| 适合持续部署 | 没有 release 分支缓冲 |
| PR 促进代码审查 | 大团队可能需要额外规范 |
| 与 CI/CD 完美集成 | 不适合维护多个生产版本 |

## 18. 常见问题解决

### 7.1 分支名冲突

**问题：**
```powershell
git checkout -b develop/gaojiFuction
# fatal: cannot lock ref 'refs/heads/develop/gaojiFuction': 'refs/heads/develop' exists
```

**原因：** Git 不允许创建以现有分支名开头的子分支。

**解决：** 使用标准前缀
```powershell
git checkout -b feature/gaojiFuction
```

### 7.2 合并冲突解决流程

```powershell
# 1. 合并时出现冲突
git merge feature/ai-player
# CONFLICT: Merge conflict in gui_game.py

# 2. 查看冲突文件
git status

# 3. 手动解决冲突（编辑文件，删除冲突标记）
# <<<<<<< HEAD
# 当前分支的内容
# =======
# 合并分支的内容
# >>>>>>> feature/ai-player

# 4. 解决后添加到暂存区
git add gui_game.py

# 5. 完成合并
git commit -m "Resolve merge conflict in gui_game.py"
```

### 7.3 合并策略

| 策略 | 命令 | 优点 | 缺点 |
|------|------|------|------|
| 快进合并 | `git merge feature` | 简单快速 | 丢失分支历史 |
| 普通合并 | `git merge --no-ff feature` | 保留分支历史 | 多一个合并提交 |
| 变基合并 | `git rebase develop` | 线性历史 | 需要 force push 风险 |
| 压缩合并 | `git merge --squash feature` | 提交历史整洁 | 丢失详细历史 |
